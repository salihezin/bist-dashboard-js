import java.util.*;
import java.time.*;

class Candle {
    double open, high, low, close, volume;
    LocalDate date;
    Candle(double o, double h, double l, double c, double v, LocalDate d) {
        open = o; high = h; low = l; close = c; volume = v; date = d;
    }
}

public class V10Strategy {

    // ===============================
    // 📊 İNDİKATÖR FONKSİYONLARI
    // ===============================
    public static List<Double> calculateLowerBand(List<Candle> candles, int length, double std) {
        List<Double> bands = new ArrayList<>();
        for (int i = length - 1; i < candles.size(); i++) {
            List<Candle> slice = candles.subList(i - length + 1, i + 1);
            double sma = slice.stream().mapToDouble(c -> c.close).average().orElse(0);
            double variance = slice.stream().mapToDouble(c -> Math.pow(c.close - sma, 2)).sum() / length;
            double stdDev = Math.sqrt(variance);
            bands.add(sma - stdDev * std);
        }
        return bands;
    }

    public static List<Double> calculateVWMA(List<Candle> candles, int period) {
        List<Double> vwma = new ArrayList<>();
        for (int i = period - 1; i < candles.size(); i++) {
            List<Candle> slice = candles.subList(i - period + 1, i + 1);
            double sumPV = slice.stream().mapToDouble(c -> c.close * c.volume).sum();
            double sumV = slice.stream().mapToDouble(c -> c.volume).sum();
            vwma.add(sumV == 0 ? 0 : sumPV / sumV);
        }
        return vwma;
    }

    // ===============================
    // 🎯 V10 GİRİŞ KOŞULU (örnek)
    // ===============================
    public static boolean checkEntrySignal(List<Candle> candles, int i, List<Double> lowerBands) {
        if (i < 3) return false;
        Candle bbTouch = candles.get(i - 3);
        Candle red = candles.get(i - 2);
        Candle green1 = candles.get(i - 1);
        Candle curr = candles.get(i);

        boolean condBB = bbTouch.low <= lowerBands.get(i - 3);
        boolean condRed = red.close < red.open;
        boolean condGreen1 = green1.close > green1.open;
        boolean condForm = (red.close < red.open && curr.close > curr.open); // örnek formasyon
        double hedefGiris = green1.close * 1.02;
        boolean condGiris = curr.close >= hedefGiris;

        return condBB && condRed && condGreen1 && condForm && condGiris;
    }
